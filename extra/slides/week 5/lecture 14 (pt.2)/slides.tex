\section{Exercise 1}
\begin{frame}{Exercise 1: Coin flipping}
\vfill
\begin{itemize}
	\item Ask the user how many coin tosses have to be performed
	\item Keep track of the number of times we get heads and tails
	\item We assume a \textit{fair coin} (i.e., unbiased 50\% change of getting head and tail at each toss)
\end{itemize}
\begin{block}{Hint}
You may use the \texttt{random.randint()} function to simulate a single coin toss.
\end{block}
\vfill
\end{frame}
%
\begin{frame}
\vfill
	\begin{table}[]
	  \begin{flushleft}
	  \begin{tabular}{ll}
	  \texttt{import random} &  \\
	  \texttt{} &  \\
	  \texttt{tail = 0} &  \\
	  \texttt{head = 0}        & \\
	  \texttt{} &  \\
	  \texttt{n = int(input("How many coin tosses? "))} &  \\
	  \texttt{for i in range(n):}        & \\
	  \texttt{\quad x = random.randint(1, 100)}        & \\
	  \texttt{\quad if x > 50:}        & \\
	  \texttt{\quad\quad head = head + 1}        & \\
	  \texttt{\quad else:}        & \\
	  \texttt{\quad\quad tail = tail + 1}        & \\
	  \texttt{} &  \\
	  \texttt{print("Number of heads: ", head)} &  \\
	  \texttt{print("Number of tails: ", tail)} &  \\
	  \end{tabular}
	  \end{flushleft}
	\end{table}
\vfill
\end{frame}

\section{Exercise 2}
\begin{frame}{Exercise 2: Sum of sequence}
\vfill
Write a function named \texttt{rolling\_sum()} that:
\begin{itemize}
	\item repeatedly asks the user to insert a number
	\item we assume integer numbers
	\item keeps track of the sum of the numbers inserted so far
	\item if the user inserts $0$, then exit and return the sum
\end{itemize}
\vfill
\end{frame}
% 
\begin{frame}
	\vfill
		\begin{table}[]
		  \begin{flushleft}
		  \begin{tabular}{ll}
		  \texttt{def rolling\_sum():} &  \\
		  \texttt{\quad this\_number = int(input("Insert a number: "))}        & \\
		  \texttt{\quad my\_sum = this\_number}        & \\
		  \texttt{} &  \\
		  \texttt{\quad while this\_number != 0:}        & \\
		  \texttt{\quad\quad this\_number = int(input("Insert a number: "))}        & \\
		  \texttt{\quad\quad my\_sum = my\_sum + this\_number} &  \\
		  \texttt{} &  \\
		  \texttt{\quad return my\_sum} &  \\
		  \texttt{} &  \\
		  \texttt{} &  \\
		  \texttt{print("Total sum is: ", rolling\_sum())} &  \\
		  \end{tabular}
		  \end{flushleft}
		\end{table}
	\vfill
\end{frame}

Write a function named \texttt{fibonacci1()} that:

	\item prints the first $n$ elements of the Fibonacci sequence.
\end{itemize}
\begin{block}{Definition}
Each number in the Fibonacci sequence is given by the sum of the two preceding ones.\\
The Fibonacci sequence starts with two `seed' numbers (\texttt{0} and \texttt{1}) and then the proper sequence starts:\\
\qquad \texttt{0, 1, }\textcolor{blue}{1, 2, 3, 5, 8, 13, 21, ...}
\end{block}

		  \texttt{} &  \\
		  \texttt{\quad print(second\_to\_last)}        & \\
		  \texttt{\quad print(last)}        & \\
		  \texttt{} &  \\
		  \texttt{\quad for i in range(n + 1):} &  \\
		  \texttt{\quad\quad new = last + second\_to\_last} &  \\
		  \texttt{\quad\quad print(new)} &  \\
		  \texttt{\quad\quad second\_to\_last = last} &  \\
		  \texttt{\quad\quad last = new  } &  \\

\section{Exercise 5}
\begin{frame}{Exercise 5: Multiples}
\vfill
Write a function named \texttt{multiples()} that:
\begin{itemize}
	\item accepts an integer number, say $n$
	\item iterates the integers in $[1, n]$
	\item for multiples of 3, print \texttt{`Mult3'} instead of the number
	\item for multiples of 5, print \texttt{`Mult5'} instead of the number
	\item for multiples of 5 and 3, print \texttt{`Mult35'} instead of the number
\end{itemize}
\vfill
\end{frame}
%
\begin{frame}
	\vfill
		\begin{table}[]
		  \begin{flushleft}
		  \begin{tabular}{ll}
		  \texttt{def multiples():} &  \\
		  \texttt{\quad for num in range(1, n + 1):}        & \\
		  \texttt{\quad\quad if num \% 3 == 0 and num \% 5 == 0:}        & \\
		  \texttt{\quad\quad\quad print("Mult35")}        & \\
		  \texttt{\quad\quad elif num \% 3 == 0:}        & \\
		  \texttt{\quad\quad\quad print("Mult3")}        & \\
		  \texttt{\quad\quad elif num \% 5 == 0:}        & \\
		  \texttt{\quad\quad\quad print("Mult5")}        & \\
		  \texttt{\quad\quad else:}        & \\
		  \texttt{\quad\quad\quad print(num)}        & \\
		  \end{tabular}

\section{Exercise 6}
\begin{frame}{Exercise 6: Print a subset}
\vfill
Write a function named \texttt{print\_subset()} that:
\begin{itemize}
	\item accepts an integer number, say $n$
	\item accepts an integer number, say $m \leq n$
	\item scans all integer numbers in range $[1, n]$, but prints all numbers in range $[1, m]$
\end{itemize}
\vfill
\end{frame}
%
\begin{frame}
	\vfill
	Solution \#1:
		\begin{table}[]
		  \begin{flushleft}
		  \begin{tabular}{ll}
		  \texttt{def print\_subset(n, m):} &  \\
		  \texttt{\quad for num in range(1, n + 1):}        & \\
		  \texttt{\quad\quad if num <= m:}        & \\
		  \texttt{\quad\quad\quad print(num)}        & \\
		  \end{tabular}
		  \end{flushleft}
		\end{table}
	\vfill
	Solution \#2:
	\begin{table}[]
		\begin{flushleft}
		\begin{tabular}{ll}
		\texttt{def print\_subset(n, m):} &  \\
		\texttt{\quad for num in range(1, n + 1):}        & \\
		\texttt{\quad\quad print(num)}        & \\
		\texttt{\quad\quad if num == m:}        & \\
		\texttt{\quad\quad\quad break}        & \\
		\end{tabular}
		\end{flushleft}
	  \end{table}
	\vfill
	\begin{block}{Stop and think}
		Under some circumstances, the second solution is more efficient than the first. What are those circumstances and why is it more efficient?
	\end{block}
	\vfill
\end{frame}

\section{Exercise 7}
\begin{frame}{Exercise 7: Primality test}
\vfill
Write a function named \texttt{smallerPrimes()} that:
\begin{itemize}
	\item accepts a positive integer number, say $n$
	\item prints all positive integer prime numbers smaller than $n$
\end{itemize}
\vfill
\begin{block}{Recall}
A \textit{prime} number is a natural number greater than 1 that has no positive divisors other than 1 and itself.
\end{block}
\begin{block}{Hint}
The simplest primality test aims at checking whether $n$ is evenly divisible (i.e., division with no remainder) by any number in range $[2, n/2]$. If so, the number is not prime.
\end{block}
\vfill
\end{frame}
%
\begin{frame}
	\vfill
		\begin{table}[]
		  \begin{flushleft}
		  \begin{tabular}{ll}
			\texttt{def is\_prime(num):}        & \\
			\texttt{\quad if num <= 1:}        & \\
			\texttt{\quad\quad	return False}        & \\
			\texttt{\quad for i in range(2, int(num/2) + 1):}        & \\
			\texttt{\quad\quad	if num \% i == 0:}        & \\
			\texttt{\quad\quad\quad		return False}        & \\
			\texttt{\quad return True}        & \\
		  	\texttt{ } &  \\
			\texttt{def smallerPrimes(n):}        & \\
			\texttt{\quad for i in range(2, n):}        & \\
			\texttt{\quad\quad if is\_prime(i):}        & \\
			\texttt{\quad\quad\quad print(i)}        & \\
		  \end{tabular}
		  \end{flushleft}
		\end{table}
	\vfill
\end{frame}

\section{Exercise 8}
\begin{frame}{Exercise 8: Pythagorean triples}
\vfill
Write a function named \texttt{pythagoreanTriples()} that:
\begin{itemize}
	\item accepts an integer positive number, say $n$
	\item prints all $x$ and $y$ pairs such that $x^2+y^2=n^2$
\end{itemize}
\vfill
\begin{block}{Recall}
The Pythagorean triple consists of three positive integers $x$, $y$ and $n$, with $x,y \leq n$, such that:
\begin{equation*}
	x^2 + y^2 = n^2
\end{equation*}
\end{block}
\vfill
\end{frame}
%
\begin{frame}
	\vfill
		\begin{table}[]
		  \begin{flushleft}
		  \begin{tabular}{ll}
			\texttt{def pythagoreanTriples(n):}        & \\
			\texttt{\quad for x in range(1, n):}        & \\
			\texttt{\quad\quad	for y in range(1, n):}        & \\
			\texttt{\quad\quad\quad if x**2 + y**2 == n**2:}        & \\
			\texttt{\quad\quad\quad\quad	print(x, y)}        & \\
		  \end{tabular}
		  \end{flushleft}
		\end{table}
	\vfill
\end{frame}
